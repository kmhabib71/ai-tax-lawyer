    validateXML(xml: string): { valid: boolean; errors: string[] } {
      // Validate against NBR schema
      const errors: string[] = [];

      // Check required fields
      if (!xml.includes('<TIN>')) errors.push('TIN is required');
      if (!xml.includes('<TotalIncome>')) errors.push('Total income is required');
      if (!xml.includes('<TaxPayable>')) errors.push('Tax payable is required');

      return { valid: errors.length === 0, errors };
    }
  }

  Phase 2: XML Submission via Browser Automation

  // XML Submission to NBR Portal
  class NBRSubmissionService {
    async submitXMLToNBR(xmlData: string, userCredentials: NBRCredentials): Promise<NBRSubmissionResult> {
      const browser = await playwright.chromium.launch();
      const page = await browser.newPage();

      try {
        // 1. Navigate to NBR portal
        await page.goto('https://www.etaxnbr.gov.bd/e-return/xml-upload');

        // 2. Handle authentication
        await this.authenticateUser(page, userCredentials);

        // 3. Upload XML file
        const xmlBuffer = Buffer.from(xmlData, 'utf-8');
        await page.setInputFiles('input[type="file"]', {
          name: 'tax_return.xml',
          mimeType: 'application/xml',
          buffer: xmlBuffer
        });

        // 4. Submit form
        await page.click('button[type="submit"]');

        // 5. Wait for ACK response
        await page.waitForSelector('.acknowledgment-number');
        const ackNumber = await page.textContent('.acknowledgment-number');

        // 6. Get confirmation details
        const confirmationDetails = await this.extractConfirmationDetails(page);

        return {
          success: true,
          ackNumber: ackNumber?.trim() || '',
          confirmationDetails,
          submissionTime: new Date().toISOString()
        };

      } catch (error) {
        return {
          success: false,
          error: error instanceof Error ? error.message : 'Unknown error',
          submissionTime: new Date().toISOString()
        };
      } finally {
        await browser.close();
      }
    }

    private async authenticateUser(page: Page, credentials: NBRCredentials) {
      // Handle NBR login process
      await page.fill('#tin', credentials.tin);
      await page.fill('#password', credentials.password);
      await page.click('#login-button');

      // Wait for successful login
      await page.waitForSelector('.dashboard', { timeout: 30000 });
    }

    private async extractConfirmationDetails(page: Page) {
      return {
        ackNumber: await page.textContent('.ack-number'),
        submissionDate: await page.textContent('.submission-date'),
        status: await page.textContent('.submission-status'),
        receiptUrl: await page.getAttribute('.receipt-download', 'href')
      };
    }
  }
